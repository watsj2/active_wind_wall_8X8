cmake_minimum_required(VERSION 3.13)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(PICO_BOARD pico2 CACHE STRING "Board type")

include(pico_sdk_import.cmake)

project(pico5_comm_report C CXX ASM)

pico_sdk_init()

add_executable(pico5_comm_report comm_report.c)

pico_enable_stdio_uart(pico5_comm_report 0)
pico_enable_stdio_usb(pico5_comm_report 1)

target_link_libraries(pico5_comm_report
    pico_stdlib
    hardware_pwm
    hardware_spi
)

pico_add_extra_outputs(pico5_comm_report)
